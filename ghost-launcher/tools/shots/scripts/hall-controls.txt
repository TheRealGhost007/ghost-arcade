#! env SHOTS_ADMIN=0 SHOTS_UNINSTALLED=gemdive,skyraid
# Phase 3: the hall by mouse. Pointing at a side machine gives it a soft
# glow and the hand; a click brings it to the middle; a click on the lit one
# drops the coin and plays it; clicks and the wheel wait while the coin
# drops; the wheel moves one machine per notch. The header's chips and
# every button of the control strip are buttons, and the strip names the
# keys or the pad's buttons, whichever was used last.
wait 30
expect selected 0
expect app chip:Play Enter normal
expect app chip:Daily D normal
shot hall-strip-keys

# A side machine: the glow and the hand. The lit one: the hand, its coin
# slots waiting. The carpet is nothing.
mouse 680 300
wait 3
expect app look:Coilrush hover
expect app cursor hand
shot hall-hover-glow
mouse 480 300
wait 3
expect app look:Blockfall hover
expect app cursor hand
shot hall-hover-lit
mouse 480 450
wait 3
expect app cursor default

# A click on the side machine brings it to the middle, when let go.
mouse 680 300

wait 2
expect selected 0
expect app look:Coilrush pressed
mup left
wait 2
expect selected 1
wait 30
expect app settled 1
expect app look:Brickburst hover
# Dragged off before letting go: nothing.
mdown left
mouse 480 450
mup left
wait 3
expect selected 1

# The lit machine: the coin drops. Until the game is up, clicks and the
# wheel do nothing; the X would still work.

wait 4

expect selected 1
expect app look:Brickburst absent
wait 60

expect launches 1
