#! env SHOTS_ADMIN=1 SHOTS_ORDER=az SHOTS_FAVS=skyraid
# Phase 4, the risk the plan names: a sorted row has two numberings, the
# place in the row and the line in games.txt. In A-Z with a favourite,
# admin's Edit, Remove and Add (and the dialog's question) all act on the
# machine the hall shows lit, and the row is rebuilt after each.
wait 30
expect app order A-Z
expect app row Skyraid,Blockfall,Brickburst,Chipforge,Crawlshot,Moondrop,Pond Kings,Rockdrift
expect app lit Skyraid
expect selected 2

# Edit on the second place opens Blockfall (catalog index 0).
tap RIGHT
wait 30
expect app lit Blockfall
tap E
wait 5
expect ui EDIT_GAME
expect profile Blockfall
tap ESC
wait 3
expect ui LIST
expect app lit Blockfall

# Remove asks about the machine lit (Chipforge, place 3) and keeps it on Keep.
tap RIGHT
tap RIGHT
wait 30
expect app lit Chipforge
expect selected 6
tap DELETE
wait 5
expect app dialogtitle Remove Chipforge?
tap ESC
wait 3
expect app games 8

# Remove Brickburst: place 2 in the row, catalog index 1 (the second line of
# games.txt). Chipforge, which now stands at place 2, is lit.
tap LEFT
wait 30
expect app lit Brickburst
expect selected 1
tap DELETE
wait 5
expect app dialogtitle Remove Brickburst?
tap RIGHT
tap ENTER
wait 5
expect app dialog none
expect app games 7
expect nofile .config/ghost-launcher/games.txt Brickburst|
expect file .config/ghost-launcher/games.txt Chipforge|
expect app row Skyraid,Blockfall,Chipforge,Crawlshot,Moondrop,Pond Kings,Rockdrift
expect app lit Chipforge
expect app place 2
expect app settled 1
shot row-admin-removed

# Edit Rockdrift's name (place 6, catalog index 2 now): the same line of
# games.txt, a new place in A-Z.
tap RIGHT
tap RIGHT
tap RIGHT
tap RIGHT
wait 30
expect app lit Rockdrift
tap E
wait 5
expect ui EDIT_GAME
expect profile Rockdrift
hold BACKSPACE 120
text Arcwalk
tap ENTER
tap ENTER
tap ENTER
tap ENTER
tap ENTER
wait 3
tap ENTER
wait 5
expect ui LIST
expect file .config/ghost-launcher/games.txt Arcwalk|~/bin/rockdrift|
expect app row Skyraid,Arcwalk,Blockfall,Chipforge,Crawlshot,Moondrop,Pond Kings
expect app lit Arcwalk
expect app place 1
# Its favourite and place stay with its program name, not its title.
tap F
wait 3
expect file .config/ghost-launcher/launcher.ini fav.rockdrift=1\n
expect app row Arcwalk,Skyraid,Blockfall,Chipforge,Crawlshot,Moondrop,Pond Kings

# Add: the new machine is lit, wherever A-Z puts it.
tap A
wait 5
expect ui ADD_GAME
text Bumper
tap ENTER
text ~/bin/bumper
tap ENTER
tap ENTER
tap ENTER
tap ENTER
wait 3
tap ENTER
wait 5
expect ui LIST
expect app games 8
expect app lit Bumper
expect app row Arcwalk,Skyraid,Blockfall,Bumper,Chipforge,Crawlshot,Moondrop,Pond Kings
expect app place 3
expect app note 
expect app status Not installed, and no folder to install it from.
wait 30
shot row-admin-added

# And Enter on a machine plays that machine (Blockfall: place 2, catalog index 0).
tap LEFT
wait 30
expect app lit Blockfall
tap ENTER
wait 70
expect launched Blockfall
gameexit
wait 3

# A program name that cannot be a key (capitals): no star, and F says why.
tap A
wait 5
text Oddball
tap ENTER
text ~/bin/Odd
tap ENTER
tap ENTER
tap ENTER
tap ENTER
wait 3
tap ENTER
wait 5
expect ui LIST
expect app lit Oddball
expect app look:Favourite disabled
tap F
wait 3
expect app note Its program name cannot be kept as a favourite.
expect app fav:Oddball 0
expect nofile .config/ghost-launcher/launcher.ini fav.odd
