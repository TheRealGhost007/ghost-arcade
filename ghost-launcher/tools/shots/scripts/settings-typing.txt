#! env SHOTS_ADMIN=0
# Typing a name in Settings > Player: a cursor (Left/Right/Home/End),
# Backspace and Delete, Ctrl+V, a counter, never '|'; Enter saves ("Saved"),
# Esc puts the old name back; letters never open screens while typing. New
# names stop at 32 characters. U and Shift+U start typing at once and come
# back where they began, as the old name screens did.
wait 30
tap TAB
wait 3
expect app focus Edit
expect app focusrow Arcade name
tap ENTER
wait 3
expect app typing 1
expect profile TRG
expect app cursor_at 3
text Ghost
wait 2
expect profile TRGGhost
tap LEFT
tap LEFT
wait 2
expect app cursor_at 6
shot settings-typing
tap HOME
tap DELETE
wait 2
expect profile RGGhost
expect app cursor_at 0
tap END
tap BACKSPACE
wait 2
expect profile RGGhos
clip Ace|X
tap CTRL V
wait 2
expect profile RGGhosAceX
# Letters (and Tab) only type here.
text lhvo
tap TAB
wait 3
expect ui SETTINGS
expect app typing 1
expect app tab Player
expect profile RGGhosAceXlhvo
shot settings-typing-letters
# Enter saves; begun with Edit, it stays in Settings.
tap ENTER
wait 3
expect app typing 0
expect ui SETTINGS
expect app toast Saved
expect app focus Edit
expect file .local/share/ghost-launcher/profiles.txt __DEFAULT__|RGGhosAceXlhvo\n
# Esc puts the old name back, and nothing is written; a second Esc leaves.
tap ENTER
wait 3
text ZZZ
tap ESC
wait 3
expect app typing 0
expect ui SETTINGS
expect profile RGGhosAceXlhvo
expect nofile .local/share/ghost-launcher/profiles.txt ZZZ
tap ESC
wait 3
expect ui LIST
expect app depth 1
