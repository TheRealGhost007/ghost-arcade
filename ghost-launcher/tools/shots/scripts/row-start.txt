#! env SHOTS_ADMIN=0 SHOTS_LAST_PLAYED=skyraid SHOTS_FAVS=crawlshot SHOTS_HIDDEN=rockdrift,brickburst SHOTS_EXTRA_GAMES=5
# Phase 4: the launcher opens on the machine played last; hidden machines
# are off the row (their scores stay); Settings > Player > Start on; and
# Settings > Machines: Row order, then every machine in your order with
# Favourite, Up, Down and Hide, all kept in launcher.ini. Five extra
# machines make the list longer than the seven rows it shows, so the wheel
# has something to scroll.
wait 30
expect app lit Skyraid
expect app row Crawlshot,Blockfall,Skyraid,Moondrop,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
expect app place 2
expect app settled 1
expect app hidden:Rockdrift 1
expect app fav:Crawlshot 1
shot row-start-last

# Start on: First machine is kept for the next start; the row stays put.
tap TAB
wait 3
expect app tab Player
tap DOWN
wait 2
expect app focusrow Start on
expect app startlast 1
shot settings-player-start-on
tap RIGHT
wait 3
expect app startlast 0
expect file .config/ghost-launcher/launcher.ini start_on=first\n
expect app toast Saved
tap LEFT
wait 3
expect file .config/ghost-launcher/launcher.ini start_on=last\n
expect app lit Skyraid

# Machines: the sixth tab.
tap SHIFT TAB
wait 3
expect app tab Machines
expect app focusrow Row order
expect app mine Crawlshot,Blockfall,Brickburst,Skyraid,Rockdrift,Moondrop,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
mouse 480 620
wait 3
shot settings-machines
tap DOWN
wait 2
expect app focusrow Crawlshot
expect app focus Unfavourite
tap DOWN
wait 2
expect app focusrow Blockfall
tap RIGHT
wait 2
expect app focus Down
shot settings-machines-focus
tap ENTER
wait 3
expect app toast Saved
expect app mine Crawlshot,Brickburst,Blockfall,Skyraid,Rockdrift,Moondrop,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
expect app focusrow Blockfall
expect app focus Down
expect file .config/ghost-launcher/launcher.ini place.blockfall=2\n
expect file .config/ghost-launcher/launcher.ini place.brickburst=1\n
expect nofile .config/ghost-launcher/games.txt place
# Hide Blockfall: off the row, the lit machine still lit.
tap RIGHT
wait 2
expect app focus Hide
tap ENTER
wait 3
expect app hidden:Blockfall 1
expect file .config/ghost-launcher/launcher.ini hide.blockfall=1\n
expect app row Crawlshot,Skyraid,Moondrop,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
expect app lit Skyraid
expect app focus Show
# Show Brickburst with a click (the second machine in the list).
click 632 314
wait 3
expect app hidden:Brickburst 0
expect nofile .config/ghost-launcher/launcher.ini hide.brickburst
expect app row Crawlshot,Brickburst,Skyraid,Moondrop,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
# Favourite Moondrop with a click (the sixth machine in the list): it joins
# the favourites at the top.
click 412 474
wait 3
expect app fav:Moondrop 1
expect app mine Crawlshot,Moondrop,Brickburst,Blockfall,Skyraid,Rockdrift,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
# Row order: the stepper's right arrow.
click 792 202
wait 3
expect app order Recently played
expect file .config/ghost-launcher/launcher.ini row_order=recent\n
mouse 480 620
wait 3
shot settings-machines-after
# The wheel scrolls the list.
mouse 480 400
wheel -5
wait 3
shot settings-machines-scrolled
tap ESC
wait 3
expect ui LIST
expect app lit Skyraid
expect app row Crawlshot,Moondrop,Brickburst,Skyraid,Chipforge,Pond Kings,Extra 1,Extra 2,Extra 3,Extra 4,Extra 5
wait 30
shot row-start-after
