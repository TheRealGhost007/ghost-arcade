#! env SHOTS_ADMIN=0 SHOTS_LAST_PLAYED=skyraid SHOTS_FAVS=lanehop SHOTS_HIDDEN=ghostmaze,coilrush
# Phase 4: the launcher opens on the machine played last; hidden machines
# are off the row (their scores stay); Settings > Player > Start on; and
# Settings > Machines: Row order, then every machine in your order with
# Favourite, Up, Down and Hide, all kept in launcher.ini.
wait 30
expect app lit Skyraid
expect app row Lanehop,Blockfall,Brickburst,Skyraid,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
expect app place 3
expect app settled 1
expect app hidden:Ghostmaze 1
expect app fav:Lanehop 1
shot row-start-last

# Start on: First machine is kept for the next start; the row stays put.
tap TAB
wait 3
expect app tab Player
tap DOWN
wait 2
expect app focusrow Start on
expect app startlast 1
shot settings-player-start-on
tap RIGHT
wait 3
expect app startlast 0
expect file .config/ghost-launcher/launcher.ini start_on=first\n
expect app toast Saved
tap LEFT
wait 3
expect file .config/ghost-launcher/launcher.ini start_on=last\n
expect app lit Skyraid

# Machines: the sixth tab.
tap SHIFT TAB
wait 3
expect app tab Machines
expect app focusrow Row order
expect app mine Lanehop,Blockfall,Coilrush,Brickburst,Skyraid,Ghostmaze,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
mouse 480 620
wait 3
shot settings-machines
tap DOWN
wait 2
expect app focusrow Lanehop
expect app focus Unfavourite
tap DOWN
wait 2
expect app focusrow Blockfall
tap RIGHT
wait 2
expect app focus Down
shot settings-machines-focus
tap ENTER
wait 3
expect app toast Saved
expect app mine Lanehop,Coilrush,Blockfall,Brickburst,Skyraid,Ghostmaze,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
expect app focusrow Blockfall
expect app focus Down
expect file .config/ghost-launcher/launcher.ini place.blockfall=2\n
expect file .config/ghost-launcher/launcher.ini place.coilrush=1\n
expect nofile .config/ghost-launcher/games.txt place
# Hide Blockfall: off the row, the lit machine still lit.
tap RIGHT
wait 2
expect app focus Hide
tap ENTER
wait 3
expect app hidden:Blockfall 1
expect file .config/ghost-launcher/launcher.ini hide.blockfall=1\n
expect app row Lanehop,Brickburst,Skyraid,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
expect app lit Skyraid
expect app focus Show
# Show Coilrush with a click (the second machine in the list).
click 632 314
wait 3
expect app hidden:Coilrush 0
expect nofile .config/ghost-launcher/launcher.ini hide.coilrush
expect app row Lanehop,Coilrush,Brickburst,Skyraid,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
# Favourite Brickburst with a click: it joins the favourites at the top.
click 412 394
wait 3
expect app fav:Brickburst 1
expect app mine Lanehop,Brickburst,Coilrush,Blockfall,Skyraid,Ghostmaze,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
# Row order: the stepper's right arrow.
click 792 202
wait 3
expect app order Recently played
expect file .config/ghost-launcher/launcher.ini row_order=recent\n
mouse 480 620
wait 3
shot settings-machines-after
# The wheel scrolls the list.
mouse 480 400
wheel -5
wait 3
shot settings-machines-scrolled
tap ESC
wait 3
expect ui LIST
expect app lit Skyraid
expect app row Lanehop,Brickburst,Coilrush,Skyraid,Rockdrift,Crawlshot,Moondrop,Gemdive,Girderclimb,Chipforge
wait 30
shot row-start-after
